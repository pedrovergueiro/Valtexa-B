# TensorFlow para que os dois não disputem a placa.
tensorflow==2.20.0

# Para treinar na GPU, instale o torch do índice da NVIDIA correspondente à sua
# placa — o wheel padrão do PyPI pode não trazer a arquitetura dela. As GPUs
# Blackwell (RTX 50xx, compute capability 12.0) precisam de cu128 ou mais novo:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cu128
#
# Confira o que o seu torch suporta com:
#   python -c "import torch; print(torch.cuda.get_arch_list())"

# arquitetura. Ele fica fora desta lista de propósito: as dependências declaradas dele
# puxam o torchvision, que pode trocar o torch da NVIDIA instalado acima. Instale sem
# dependências (o que ele usa de fato já está aqui, mais o scipy):
#
#   pip install --no-deps "dnaty>=2.1.1" scipy
